# สรุปการพัฒนา Authentication ถึง Middleware

## ภาพรวมการพัฒนา
เอกสารนี้สรุปการสร้างและแก้ไขไฟล์ตั้งแต่การพัฒนาระบบ Authentication จนถึง Middleware รวมถึงการปรับปรุงจาก fetch API เป็น service patterns

### 1.4 สร้าง Auth Controller (`server/controllers/auth.controllers.js`)
```javascript
import db from "../model/index.js";
const User = db.User;
const Role = db.Role;
import config from "../config/auth.config.js";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import { Op } from "sequelize";

const authController = {};

// สมัครสมาชิก
authController.register = async (req, res) => {
  const { username, name, email, password } = req.body;
  
  if (!username || !name || !email || !password) {
    res.status(400).send({ message: "ข้อมูลไม่ครบถ้วน!" });
    return;
  }

  await User.findOne({ where: { username } }).then((user) => {
    if (user) {
      res.status(400).send({ message: "Username ถูกใช้แล้ว!" });
      return;
    }

    const newUser = {
      username,
      name,
      email,
      password: bcrypt.hashSync(password, 8)
    };
    
    User.create(newUser)
      .then((user) => {
        if (req.body.roles) {
          Role.findAll({
            where: { name: { [Op.or]: req.body.roles } },
          }).then((roles) => {
            user.setRoles(roles).then(() => {
              res.send({ message: "สมัครสมาชิกสำเร็จ!" });
            });
          });
        } else {
          user.setRoles([1]).then(() => {
            res.send({ message: "สมัครสมาชิกสำเร็จ!" });
          });
        }
      });
  });
};

// เข้าสู่ระบบ
authController.login = async (req, res) => {
  const { username, password } = req.body;
  
  if (!username || !password) {
    res.status(400).send({ message: "กรุณากรอก Username และ Password!" });
    return;
  }

  await User.findOne({ where: { username } })
  .then((user) => {
    if (!user) {
      res.status(404).send({ message: "ไม่พบผู้ใช้!" });
      return;
    }
    
    const passwordIsValid = bcrypt.compareSync(password, user.password);
    if (!passwordIsValid) {
      res.status(401).send({ message: "รหัสผ่านไม่ถูกต้อง!" });
      return;
    }
    
    const token = jwt.sign({ id: user.id, username: user.username }, config.secret, {
      expiresIn: 86400, // 24 ชั่วโมง
    });
    
    const authorities = [];
    user.getRoles().then((roles) => {
      for(let i = 0; i < roles.length; i++) {
        authorities.push("ROLE_" + roles[i].name.toUpperCase());
      }
      res.status(200).send({
        accessToken: token,
        id: user.id,
        username: user.username,
        name: user.name,
        email: user.email,
        authorities: authorities,
        roleId: roles[0]?.id ?? null,
      });
    });
  });
};

export default authController;
```

## 2. การสร้าง Service Layer (แทนที่ fetch API)

### 2.1 สร้าง API Instance (`client/src/service/api.js`)
```javascript
import axios from 'axios';
import tokenService from './token.service';

const baseURL = import.meta.env.VITE_BASE_URL;

const instance = axios.create({
    baseURL: baseURL,
    headers: {
        'Content-Type': 'application/json',
    },
});

// เพิ่ม interceptors สำหรับ request
instance.interceptors.request.use(
    (config) => {
        const token = tokenService.getlocalAccessToken();
        if (token) {
            config.headers['Authorization'] = `Bearer ${token}`;
        }
        return config;
    },
    (error) => {
        return Promise.reject(error);
    }
);

export default instance;
```

### 2.2 สร้าง Token Service (`client/src/service/token.service.js`)
```javascript
const getlocalAccessToken = () => {
    const user = JSON.parse(localStorage.getItem("user"));
    return user?.accessToken;
};

const updateLocalAccessToken = (token) => {
    let user = JSON.parse(localStorage.getItem("user"));
    user.accessToken = token;
    localStorage.setItem("user", JSON.stringify(user));
};

const getUser = () => {
    return JSON.parse(localStorage.getItem("user"));
};

const setUser = (user) => {
    localStorage.setItem("user", JSON.stringify(user));
};

const removeUser = () => {
    localStorage.removeItem("user");
};

const tokenService = {
    getlocalAccessToken,
    updateLocalAccessToken,
    getUser,
    setUser,
    removeUser,
};

export default tokenService;
```

### 2.3 สร้าง Auth Service (`client/src/service/auth.service.js`)
```javascript
import api from './api.js';
import tokenService from './token.service.js';

const AUTH_API = '/auth';

const register = async (username, name, email, password) => {
    return await api.post(`${AUTH_API}/signup`, {username, name, email, password});
};

const login = async (username, password) => {
    const response = await api.post(`${AUTH_API}/signin`, {username, password});
    
    // บันทึกข้อมูลผู้ใช้ลง localStorage
    if (response.data.accessToken) {
        tokenService.setUser(response.data);
    }
    
    return response;
};

const logout = () => {
    tokenService.removeUser();
};

const authService = {
    register,
    login,
    logout
};

export default authService;
```

## 4. การสร้าง Auth Context

### 4.1 สร้าง AuthContext (`client/src/context/authcontext.js`)
```javascript
import React, { useState, useContext, createContext, useEffect } from "react";
import authService from "../service/auth.service";
import tokenService from "../service/token.service";

const AuthContext = createContext(null);

export const AuthProvider = ({ children }) => {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true);

    const login = (userData) => {
        setUser(userData);
        tokenService.setUser(userData);
    };

    const logout = () => {
        authService.logout();
        setUser(null);
        tokenService.removeUser();
    };

    const getUser = () => {
        return tokenService.getUser();
    };

    // เริ่มต้นข้อมูลผู้ใช้จาก localStorage
    useEffect(() => {
        const storedUser = tokenService.getUser();
        if (storedUser && storedUser.accessToken) {
            setUser(storedUser);
        }
        setLoading(false);
    }, []);

    const value = {
        user,
        login,
        logout,
        getUser,
        loading
    };

    return (
        <AuthContext.Provider value={value}>
            {children}
        </AuthContext.Provider>
    );
};

export const useAuthContext = () => {
    const context = useContext(AuthContext);
    if (!context) {
        throw new Error('useAuthContext must be used within AuthProvider');
    }
    return context;
};
```

## 5. การสร้าง Protection Components

### 5.1 สร้าง AdminPage.jsx
```javascript
import React from 'react';
import { Navigate } from 'react-router-dom';
import { useAuthContext } from '../context/authcontext';
import tokenService from '../service/token.service';

const AdminPage = ({ children }) => {
    const { user, loading } = useAuthContext();
    
    // ถ้าถังโหลด รอก่อน
    if (loading) {
        return <div>Loading...</div>;
    }
    
    // ตรวจสอบจาก localStorage ถ้า context ยังไม่โหลด
    const currentUser = user || tokenService.getUser();
    
    if (!currentUser) {
        return <Navigate to="/login" replace />;
    }
    
    const hasAdminRole = currentUser.authorities?.includes('ROLE_ADMIN');
    
    if (!hasAdminRole) {
        return <Navigate to="/not-allowed" replace />;
    }
    
    return children;
};

export default AdminPage;
```

## 6. การสร้าง Middleware

### 6.1 สร้าง JWT Middleware (`server/middleware/authjwt.js`)
```javascript
import jwt from "jsonwebtoken";
import config from "../config/auth.config.js";

const verifytoken = (req, res, next) => {
    let token = req.headers["authorization"];
    
    if (!token) {
        return res.status(403).send({ message: "ไม่มี Token!" });
    }
    
    // แยก token จาก "Bearer TOKEN_VALUE"
    if (token.startsWith('Bearer ')) {
        token = token.slice(7); // เอา "Bearer " ออก
    }
    
    jwt.verify(token, config.secret, (err, decoded) => {
        if (err) {
            return res.status(401).send({ message: "ไม่ได้รับอนุญาต!" });
        }
        req.userId = decoded.id;
        next();
    });
}

const authJwt = { verifytoken };

export default authJwt;
```

## 7. การแก้ไข Model Relationships

### 7.1 แก้ไข Model Index (`server/model/index.js`)
```javascript
import sequelize from "../model/db.js";
import { Sequelize } from "sequelize";
import User from "./user.model.js";
import Role from "./role.model.js";

const db = {};
db.sequelize = sequelize;
db.Sequelize = Sequelize;
db.User = User;
db.Role = Role;

// ความสัมพันธ์ - เชื่อม Users และ Roles ผ่านตาราง user_roles
db.User.belongsToMany(db.Role, {
    through: "user_roles",
    foreignKey: "userUsername",  // Foreign key ในตาราง user_roles
    otherKey: "roleId",          // Foreign key อื่นในตาราง user_roles
    as: "roles"                  // ชื่อแทนสำหรับ query ที่ง่ายขึ้น
});

db.Role.belongsToMany(db.User, {
    through: "user_roles", 
    foreignKey: "roleId",        // Foreign key ในตาราง user_roles
    otherKey: "userUsername",    // Foreign key อื่นในตาราง user_roles
    as: "users"                  // ชื่อแทนสำหรับ query ที่ง่ายขึ้น
});

export default db;
```

## 8. การแก้ไข Card Component สำหรับ Role-Based UI

### 8.1 แก้ไข Card.jsx
**เพิ่มการตรวจสอบสิทธิ์:**
```javascript
import tokenService from '../service/token.service';

const Card = ({ restaurant }) => {
    const user = tokenService.getUser();
    const authorities = user?.authorities || [];
    
    // ตรวจสอบสิทธิ์สำหรับปุ่มแก้ไข/ลบ
    const showEdit = authorities.includes('ROLE_ADMIN') || authorities.includes('ROLE_MODERATOR');
    const showDelete = authorities.includes('ROLE_ADMIN') || authorities.includes('ROLE_MODERATOR');

    const handleEdit = () => {
        window.location.href = `/update/${restaurant.id}`;
    };

    const handleDelete = async () => {
        const result = await Swal.fire({
            title: 'คุณแน่ใจหรือไม่?',
            text: "คุณต้องการลบร้านอาหารนี้?",
            icon: 'warning',
            showCancelButton: true,
            confirmButtonColor: '#d33',
            cancelButtonColor: '#3085d6',
            confirmButtonText: 'ใช่, ลบเลย!',
            cancelButtonText: 'ยกเลิก'
        });

        if (result.isConfirmed) {
            try {
                await restaurantService.deleteRestaurantById(restaurant.id);
                Swal.fire('ลบแล้ว!', 'ร้านอาหารถูกลบแล้ว', 'success');
                window.location.reload();
            } catch (error) {
                Swal.fire('ข้อผิดพลาด!', 'ไม่สามารถลบร้านอาหารได้', 'error');
            }
        }
    };

    return (
        <div className="card">
            {/* เนื้อหาการ์ด */}
            
            {/* ปุ่มแก้ไข/ลบ - แสดงเฉพาะผู้ที่มีสิทธิ์ */}
            <div className="card-actions">
                {showEdit && (
                    <button onClick={handleEdit} className="btn btn-primary">
                        แก้ไข
                    </button>
                )}
                {showDelete && (
                    <button onClick={handleDelete} className="btn btn-error">
                        ลบ
                    </button>
                )}
            </div>
        </div>
    );
};
```
